#include <tunables/global>

# AppArmor profile for the Vicohome MQTT Bridge add-on. Supervisor loads this file and
# confines the container with it. It allows only what the bridge uses: Python, ffmpeg and
# ffprobe, the add-on's own folders, and IP sockets for MQTT, the cloud API and the camera's
# WebRTC (SD card) link. Anything else is denied and logged in the host's kernel log.
profile vicohome_mqtt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Network: MQTT and HTTPS over TCP, WebRTC/ICE over UDP, netlink to list the host's
  # interfaces (aioice), and Unix socket pairs (asyncio's internal wake-up pipe).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
  network unix stream,
  network unix dgram,

  # Stop/restart from Supervisor, and the bridge ending its own ffmpeg children on a timeout.
  signal (receive),
  signal (send) peer=@{profile_name},

  # Docker's init (Supervisor starts add-ons with --init) and the Python runtime.
  /sbin/docker-init rix,
  /usr/bin/tini rix,
  /usr/local/bin/ r,
  /usr/local/bin/python3* rix,
  /usr/local/lib/** rm,
  /app/ r,
  /app/** r,

  # Clips and snapshots. ldconfig is run by ctypes.util.find_library (aioice's interface list).
  /usr/bin/ffmpeg rix,
  /usr/bin/ffprobe rix,
  /{usr/,}sbin/ldconfig{,.real} rix,

  # Read-only system files: DNS, TLS certificates, time zones, shared libraries.
  /etc/** r,
  /usr/share/** r,
  /usr/lib/** rm,

  # Process and CPU information Python, OpenSSL and ffmpeg read about themselves.
  owner @{PROC}/@{pid}/** r,
  @{PROC}/cpuinfo r,
  @{PROC}/meminfo r,
  @{PROC}/stat r,
  @{PROC}/sys/crypto/fips_enabled r,
  @{PROC}/sys/kernel/random/* r,
  /sys/devices/system/cpu/ r,
  /sys/devices/system/cpu/** r,
  /sys/fs/cgroup/cpu.max r,

  /dev/null rw,
  /dev/urandom r,
  /dev/random r,

  # Scratch space (SD clip parts, ffmpeg playlists).
  /tmp/ rw,
  /tmp/** rwk,

  # Add-on options, login token, visit state and keys.
  /data/ r,
  /data/** rwk,

  # Snapshots: /media/<media_folder>/ (map: media).
  /media/ r,
  /media/** rw,

  # Clips: <config>/www/<clip_folder>/, served at /local/ (map: homeassistant_config).
  # Only www is writable; the rest of the Home Assistant config is not readable.
  /homeassistant/ r,
  /homeassistant/configuration.yaml r,
  /homeassistant/www/ rw,
  /homeassistant/www/** rw,
  /config/ r,
  /config/configuration.yaml r,
  /config/www/ rw,
  /config/www/** rw,
}
